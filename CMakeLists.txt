cmake_minimum_required(VERSION 3.13)
project(resolve_aac_bridge VERSION 0.2.0 LANGUAGES CXX ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "This runtime bridge targets Linux x86-64 only")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RESOLVE_SDK "" CACHE PATH "Path to CodecPlugin/Examples/x264_encoder_plugin")
set(FDK_ROOT "" CACHE PATH "FDK-AAC static installation prefix")
if(NOT EXISTS "${RESOLVE_SDK}/wrapper/plugin_api.cpp")
  message(FATAL_ERROR "Set RESOLVE_SDK to the codec SDK shipped with Resolve")
endif()
find_path(FDK_INCLUDE fdk-aac/aacdecoder_lib.h HINTS "${FDK_ROOT}/include" REQUIRED)
find_library(FDK_LIBRARY NAMES libfdk-aac.a HINTS "${FDK_ROOT}/lib" "${FDK_ROOT}/lib64" REQUIRED)
find_package(Threads REQUIRED)
set(SDK_SOURCES "${RESOLVE_SDK}/wrapper/host_api.cpp" "${RESOLVE_SDK}/wrapper/plugin_api.cpp")
add_library(aac_decoder_probe SHARED src/decoder.cpp src/bridge.cpp src/bridge.S ${SDK_SOURCES})
target_include_directories(aac_decoder_probe PRIVATE src "${RESOLVE_SDK}" "${RESOLVE_SDK}/include" "${FDK_INCLUDE}")
target_compile_definitions(aac_decoder_probe PRIVATE AAC_RUNTIME_BRIDGE)
target_compile_options(aac_decoder_probe PRIVATE -fvisibility=hidden)
target_link_libraries(aac_decoder_probe PRIVATE "${FDK_LIBRARY}" Threads::Threads m ${CMAKE_DL_LIBS})
target_link_options(aac_decoder_probe PRIVATE -Wl,-z,defs -Wl,-Bsymbolic -Wl,--exclude-libs,ALL)
set_target_properties(aac_decoder_probe PROPERTIES PREFIX "" SUFFIX ".dvcp")
install(TARGETS aac_decoder_probe LIBRARY DESTINATION aac_decoder_probe.dvcp.bundle/Contents/Linux-x86-64)
add_executable(check-resolve tests/scan.cpp src/bridge.cpp src/bridge.S)
target_compile_options(check-resolve PRIVATE -UNDEBUG)
target_include_directories(check-resolve PRIVATE src)
target_link_libraries(check-resolve PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS check-resolve RUNTIME DESTINATION tools)
enable_testing()
add_executable(config-test tests/config_test.cpp)
target_compile_options(config-test PRIVATE -UNDEBUG)
target_include_directories(config-test PRIVATE src)
add_test(NAME aac-metadata COMMAND config-test)
add_executable(decoder-test tests/decoder_test.cpp ${SDK_SOURCES})
target_compile_options(decoder-test PRIVATE -UNDEBUG)
target_include_directories(decoder-test PRIVATE src "${RESOLVE_SDK}" "${RESOLVE_SDK}/include" "${FDK_INCLUDE}")
target_link_libraries(decoder-test PRIVATE "${FDK_LIBRARY}" Threads::Threads m)
add_test(NAME aac-roundtrip COMMAND decoder-test)
